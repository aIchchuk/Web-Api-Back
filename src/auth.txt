🔗 1. Register/Login Request from Frontend
Inside your login or register page, you’ll call your backend like this:

🧪 Example: Login Request (React)
js
Copy
Edit
import { useContext, useState } from "react";
import { AuthContext } from "../AuthContextProvider";
import axios from "axios";

const LoginPage = () => {
  const { login } = useContext(AuthContext);
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const handleLogin = async () => {
    try {
      const response = await axios.post("http://localhost:5000/auth/login", {
        email,
        password,
      });

      const { token, user } = response.data;
      login(user, token); // this updates the context
    } catch (err) {
      console.error("Login failed:", err.response?.data?.message || err.message);
    }
  };

  return (
    <div>
      <input placeholder="Email" onChange={e => setEmail(e.target.value)} />
      <input type="password" placeholder="Password" onChange={e => setPassword(e.target.value)} />
      <button onClick={handleLogin}>Log In</button>
    </div>
  );
};

🛡️ 4. Using Token for Protected Routes
When making requests to routes like /playlist, /song, etc., pass the token:

js
Copy
Edit
const token = localStorage.getItem("token");

const response = await axios.get("http://localhost:5000/song", {
  headers: {
    Authorization: `Bearer ${token}`,
  },
});
Make sure your backend uses JWT auth on those protected routes.

🔁 5. (Optional) Add Middleware to Protect Routes
On backend routes like /playlist, add JWT auth middleware:

js
Copy
Edit
import jwt from "jsonwebtoken";

export const verifyToken = (req, res, next) => {
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    return res.status(401).json({ message: "Unauthorized" });
  }

  const token = authHeader.split(" ")[1];

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded; // now you can use req.user in route
    next();
  } catch (err) {
    res.status(401).json({ message: "Invalid token" });
  }
};
Use it in routes like:

js
Copy
Edit
router.get("/playlist", verifyToken, (req, res) => {
  // Only logged-in users can access this
});
